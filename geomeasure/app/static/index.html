Python Automated Ingestion:

import requests # 1. Upload & Measure file with open("sample_polygon.kml", "rb") as f: upload_res = requests.post("http://localhost:8000/api/files/", files={"file": f}).json() file_id = upload_res["id"] print(f"File processed: {file_id}, Status: {upload_res['status']}") # 2. Fetch full measurements and provenance measurements = requests.get(f"http://localhost:8000/api/files/{file_id}/measurements/").json() for feat in measurements["features"]: print(f"[{feat['geometry_type']}] Value: {feat['measurement_value']} {feat['measurement_unit']} | Status: {feat['measurement_status']}") print(f"Provenance: {feat['provenance']['method_description']}")
